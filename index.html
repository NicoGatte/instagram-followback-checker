Chi non ti segue su Instagram

Nessun login richiesto — funziona con uno script nella console del browser

1
Apri Instagram nel browser e fai login

Vai su instagram.com da Chrome, Firefox o Edge (non da app). Devi essere loggato.

2
Apri la console del browser

Premi F12 (oppure tasto destro → Ispeziona), poi clicca sulla tab Console.

3
Incolla questo script e premi Invio

Lo script usa la tua sessione già attiva per chiamare le API interne di Instagram. Nessun dato viene inviato fuori dal tuo browser.

(async () => {
  const sleep = ms => new Promise(r => setTimeout(r, ms));
  const csrf = document.cookie.split(';').find(c => c.trim().startsWith('csrftoken='))?.split('=')[1];
  const headers = { 'x-ig-app-id': '936619743392459', 'x-csrftoken': csrf };

  const username = prompt("Il tuo username Instagram (senza @):");
  const userResp = await fetch(`/api/v1/users/web_profile_info/?username=${username}`, { headers });
  const userData = await userResp.json();
  const userId = userData.data.user.id;
  console.log('ID trovato:', userId);

  async function fetchAll(type) {
    let list = [], cursor = null;
    while (true) {
      const url = `/api/v1/friendships/${userId}/${type}/?count=200` + (cursor ? `&max_id=${cursor}` : '');
      const d = await (await fetch(url, { headers })).json();
      if (!d.users) break;
      list.push(...d.users.map(u => u.username));
      console.log(`${type}: ${list.length}`);
      if (!d.next_max_id) break;
      cursor = d.next_max_id;
      await sleep(1500);
    }
    return list;
  }

  const followers = await fetchAll('followers');
  const following = await fetchAll('following');
  const nonSeguono = following.filter(u => !followers.includes(u));
  const nonRicambiati = followers.filter(u => !following.includes(u));
  const output = JSON.stringify({ followers: followers.length, following: following.length, nonSeguono, nonRicambiati });
  console.log('%cCOPIA QUESTO:', 'color:lime;font-weight:bold;font-size:14px');
  console.log(output);
  try { copy(output); } catch(e) {}
  alert(`Fatto! ${nonSeguono.length} non ti seguono, ${nonRicambiati.length} non ricambiati. Output copiato.`);
})();
